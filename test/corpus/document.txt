// A small report template, written to exercise the grammar.
#let report(title: "", authors: (), body) = {
  set document(title: title, author: authors)
  set page(
    paper: "a4",
    margin: (x: 2.5cm, y: 3cm), // generous margins
    numbering: "1",
  )
  set text(font: ("Libertinus Serif", "Noto Serif CJK SC"), lang: "zh")
  show heading.where(level: 1): it => {
    pagebreak(weak: true)
    block(below: 1em)[
      #set text(size: 18pt)
      #smallcaps(it.body)
    ]
  }
  show link: underline
  show "排版": word => text(fill: blue, word)

  align(center, text(17pt, strong(title)))
  for (i, name) in authors.enumerate() {
    if i > 0 [, ] else [By ]
    name
  }
  body
}

#show: report.with(
  title: "排版笔记",
  authors: ("甲", "乙"), /* two authors */
